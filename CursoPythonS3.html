'''FUNCIONES**************************************************************

     Se utiliza para organizar el programa en partes más pequeñas, reutilizables y 
     fáciles de mantener. En lugar de repetir el mismo código varias veces, una 
     función permite ejecutar una acción con solo llamarla por su nombre.

        🔹 Recibe datos de entrada (parámetros o argumentos).
        🔹 Realiza un proceso interno con esos datos.

        🔹 Puede reutilizarse en distintas partes del programa.

'''MODULARIDAD**************************************************************
 📚 La modularidad consiste en dividir un programa grande en partes más pequeñas 
     llamadas **módulos**, donde cada uno cumple una función específica dentro 
     del sistema. Esta práctica mejora la organización, la comprensión y el 
     mantenimiento del código.

 🧠  Ventajas de la modularidad:
        🔹 Permite mantener un código más **ordenado y legible**, separando las 
            responsabilidades de cada parte del programa.
        🔹 Facilita la **reutilización de código**, ya que un módulo puede usarse 
            en diferentes proyectos o programas.
        🔹 Favorece el **trabajo en equipo**, porque cada persona puede trabajar 
            en un módulo distinto sin interferir con los demás.
        🔹 Mejora la **depuración y actualización**, ya que se pueden modificar 
            partes del código sin afectar todo el programa.
        🔹 En **econometría**, la modularidad permite desarrollar bibliotecas 
            personalizadas para cálculos estadísticos, funciones de regresión o 
            manipulación de datos, que luego pueden importarse en distintos análisis.

 💡  En Python, un módulo es simplemente un archivo con extensión `.py` que 
      contiene funciones, clases o variables que pueden ser utilizadas en otros 
      programas mediante la instrucción `import`.

 ✍️  Pasos para aplicar la modularidad:
        1️⃣ Crear un archivo con las funciones o herramientas necesarias (por ejemplo: `mimodulo.py`).
        2️⃣ Importar ese módulo en otro archivo para reutilizar las funciones.
        3️⃣ Ejecutar el código llamando las funciones del módulo según sea necesario.
'''

'''LISTAS**************************************************************
 📚 Una **lista** es una colección ordenada y modificable que permite almacenar

        🔹 Son **ordenadas**, es decir, cada elemento tiene una posición (índice).  
        🔹 Son **mutables**, se pueden modificar, agregar o eliminar elementos.  
        🔹 Admiten **tipos de datos mixtos**, como texto, números o incluso otras listas.  
        🔹 Se declaran entre corchetes `[]`.  

 💡  Importancia:
        - Permiten organizar grandes volúmenes de datos de forma estructurada.  
        - Son ideales para manejar **series de observaciones** o **variables económicas** 
          en proyectos de análisis y econometría.  
        - Desarrollan la **lógica iterativa**, ya que pueden recorrerse con bucles para

import os
def mostrar_menu():
    os.system("cls")
    print("\n--- MENÚ DE INVENTARIO ---")
    print("1 - Agregar Producto")
    print("2 - Listar Productos")
    print("3 - Modificar un Producto")
    print("4 - Eliminar un Producto")
    print("5 - Salir")

def agregar_producto(lista):
    producto = input("Ingrese el nombre del producto: ")
    lista.append(producto)
    print(f"✅ '{producto}' agregado con éxito.")    

def listar_productos(lista):
    if len(lista) == 0:
        print("⚠️ No hay productos en la lista.")
    else:
        print("\n📋 Lista de Productos:")
        for i, producto in enumerate(lista, start=1):
            print(f"{i}. {producto}")
    
def modificar_producto(lista):
    listar_productos(lista)
    if len(lista) == 0:
        return
    try:
        indice = int(input("Ingrese el número del producto a modificar: "))
        if 1 <= indice <= len(lista):
            nuevo = input("Ingrese el nuevo nombre: ")
            lista[indice - 1] = nuevo
            print("✏️ Producto modificado correctamente.")
        else:
            print("❌ Número inválido.")
    except ValueError:
        print("⚠️ Debe ingresar un número válido.")    

def eliminar_producto(lista):
    listar_productos(lista)
    if len(lista) == 0:
        return
    try:
        indice = int(input("Ingrese el número del producto a eliminar: "))
        if 1 <= indice <= len(lista):
            eliminado = lista.pop(indice - 1)
            print(f"🗑️ '{eliminado}' eliminado correctamente.")
        else:
            print("❌ Número inválido.")
    except ValueError:
        print("⚠️ Debe ingresar un número válido.")

            listar_productos(productos)
        elif opcion == "3":
            modificar_producto(productos)
        elif opcion == "4":
            eliminar_producto(productos)

'''TUPLAS**************************************************************
 📚 Una **tupla** es una colección ordenada de elementos, similar a una lista, 
     pero con la diferencia de que **no se puede modificar después de su creación**. 
     Esto significa que una vez definida, no se pueden agregar, eliminar ni cambiar 
     elementos. 

 🧠  Características principales:
        🔹 Son **ordenadas** → cada elemento tiene una posición (índice).  
        🔹 Son **inmutables** → su contenido no puede alterarse.  
        🔹 Admiten **tipos de datos mixtos** (números, textos, listas, etc.).  
        🔹 Se declaran entre paréntesis `( )`.  
        🔹 Pueden contener un solo valor, pero debe terminar en coma.  

 💡  Importancia:
        - Aumentan la **seguridad de los datos**, evitando cambios accidentales.  
        - Son más **eficientes** en memoria que las listas.  
        - Útiles cuando se necesita almacenar información fija o constante.  
        - En **econometría**, las tuplas pueden representar **observaciones estáticas**, 
          como parámetros de un modelo o coordenadas de una zona geográfica.  

 ✍️  Sintaxis:       
'''
#Declaracion
categorias = ("Electrónica", "Ropa", "Hogar")
una_sola = ("Única",)  # atención a la coma

#Acceso
categorias[0]  # "Electrónica"
# desempaquetado
a, b, c = categorias
#Al ser inmutables, no tienen append, remove, sort, etc.


'''
EJEMPLO:Registro de parámetros fijos en un modelo energético
'''
# Tupla con parámetros fijos de una planta solar (capacidad, eficiencia, costo_unitario)
planta_solar = ("Planta Solar El Zamorano", 2500, 0.85, 0.12)
# Desempaquetado de valores
nombre, capacidad_kw, eficiencia, costo_kwh = planta_solar
# Cálculo del costo estimado de generación anual
produccion_anual_kwh = capacidad_kw * eficiencia * 365 * 24
costo_total_anual = produccion_anual_kwh * costo_kwh
# Resultados
print(f"📍 {nombre}")
print(f"Capacidad instalada: {capacidad_kw} kW")
print(f"Eficiencia promedio: {eficiencia * 100:.1f}%")
print(f"Costo estimado anual: ${costo_total_anual:,.2f}")
'''
🎓 Justificación: ¿Por qué usar tuplas en lugar de otras colecciones?
Inmutabilidad y seguridad de los datos:Los parámetros de una planta (como la capacidad o la eficiencia 
nominal) no deben cambiar durante la ejecución del programa. Usar una tupla garantiza que estos valores 
permanezcan fijos e intactos, evitando errores humanos o modificaciones accidentales.

Eficiencia en memoria y rendimiento:Las tuplas son más ligeras que las listas, ya que Python las 
almacena en estructuras más compactas y no reserva espacio adicional para modificaciones. 
En análisis de datos grandes —por ejemplo, modelos energéticos con miles de observaciones— 
esto mejora el rendimiento.

Representación lógica de “constantes agrupadas”:
Una tupla comunica la idea de un registro estático o entidad fija, muy útil para parámetros de calibración,
coordenadas geográficas o configuraciones iniciales. En cambio, una lista se usaría para datos dinámicos 
o editables, como consumos mensuales o precios históricos.

Mayor integridad conceptual:
Desde la perspectiva de diseño de software y análisis econométrico, las tuplas refuerzan la integridad de 
los supuestos del modelo, al separar claramente los valores que se calculan (variables) de los que se 
establecen como fijos (constantes o condiciones iniciales).
'''

'''CONJUNTOS**************************************************************
 📚 Un **conjunto** (set) es una colección **no ordenada** y **sin elementos duplicados**.  
     Se utiliza para almacenar valores únicos y realizar operaciones matemáticas 
     como unión, intersección y diferencia.  

 🧠  Características principales:
        🔹 No mantiene un orden fijo (no tiene índice).  
        🔹 No permite elementos repetidos.  
        🔹 Es **mutable**, es decir, se pueden agregar o eliminar elementos.  
        🔹 Se define con llaves `{}` o usando la función `set()`.  

 💡  Importancia:
        - Permiten **eliminar duplicados** automáticamente de un conjunto de datos.  
        - Facilitan operaciones **comparativas** entre grupos de información.  
        - Son muy útiles para análisis donde se necesitan identificar elementos 
          comunes o exclusivos entre categorías o periodos.  
        - En **econometría o análisis energético**, pueden emplearse para comparar 
          listas de zonas, clientes o registros entre dos periodos de medición.  

 ✍️  Sintaxis:
'''
#Declaracion
clientes = {"Cliente1", "Cliente2", "Cliente3"}
vacio = set()

#Operaciones
clientes.add("Cliente4")                  # agrega 1
clientes.update(["Cliente2", "Cliente5"]) # agrega varios
clientes.discard("ClienteX")              # no falla si no existe
clientes.remove("Cliente1")               # falla si no existe
extraido = clientes.pop()                 # saca un elemento arbitrario
clientes.clear()                          # vacía el set

#operaciones de conjuntos 
a = {"A", "B", "C"}
b = {"B", "C", "D"}

a | b   # unión -> {"A","B","C","D"}
a & b   # intersección -> {"B","C"}
a - b   # diferencia -> {"A"}
a ^ b   # diferencia simétrica -> {"A","D"}

# Los conjuntos no tienen índice; se recorren con for
for elemento in a:
    print("Elemento:", elemento)
    
'''⚡ Ejemplo: Comparación de clientes activos entre dos periodos de facturación eléctrica'''
# Conjuntos de clientes por mes
clientes_mes1 = {"Ana", "Carlos", "Elena", "David", "Fernanda", "Luis"}
clientes_mes2 = {"Carlos", "Elena", "Luis", "Gabriela", "Raúl"}

# Operaciones de conjuntos
clientes_continuan = clientes_mes1 & clientes_mes2       # intersección
clientes_nuevos = clientes_mes2 - clientes_mes1          # diferencia
clientes_perdidos = clientes_mes1 - clientes_mes2        # diferencia inversa
total_unico = clientes_mes1 | clientes_mes2              # unión sin duplicados

# Resultados
print("🔸 Clientes que continúan:", clientes_continuan)
print("🟢 Clientes nuevos:", clientes_nuevos)
print("🔴 Clientes perdidos:", clientes_perdidos)
print("📊 Total único de clientes:", total_unico)

'''SALIDA
🔸 Clientes que continúan: {'Carlos', 'Elena', 'Luis'}
🟢 Clientes nuevos: {'Gabriela', 'Raúl'}
🔴 Clientes perdidos: {'Ana', 'David', 'Fernanda'}
📊 Total único de clientes: {'Carlos', 'Ana', 'Luis', 'Elena', 'Raúl', 'David', 'Fernanda', 'Gabriela'}
'''

'''🎓 Justificación: ¿Por qué usar conjuntos y no listas, tuplas o diccionarios?
Eliminación automática de duplicados:
En contextos empresariales o energéticos, puede haber registros repetidos por errores de captura o 
consolidación de bases de datos. Los conjuntos eliminan duplicados de forma automática, garantizando 
que cada cliente o registro se contabilice solo una vez.
👉 En cambio, una lista permitiría duplicados y requeriría pasos adicionales (como convertir a set() 
   o usar filtros).

Operaciones matemáticas rápidas y expresivas:
Los conjuntos permiten realizar comparaciones directas entre grupos (unión, intersección, diferencia) 
con una sintaxis clara y eficiente.
Estas operaciones son más rápidas que recorrer listas con bucles, especialmente en grandes volúmenes 
de datos, como análisis de clientes entre dos meses de facturación.

Desempeño y eficiencia computacional:
Los set están implementados con tablas hash, lo que hace que operaciones como búsqueda o verificación (in) 
sean mucho más rápidas que en listas o tuplas, donde se requiere recorrer todos los elementos.

Claridad semántica y seguridad lógica:
Desde el punto de vista del diseño del programa, los conjuntos expresan la idea de pertenencia y 
comparación, no de orden o secuencia.
Por ejemplo, “¿qué clientes se mantuvieron?” o “¿cuáles son exclusivos del mes 2?” son preguntas de 
conjunto, no de índice ni de posición.

Flexibilidad controlada:
A diferencia de las tuplas (inmutables) y las listas (ordenadas), los conjuntos son mutables y 
no ordenados, lo que permite agregar o eliminar elementos según cambien los datos, sin preocuparse por 
el orden.
'''

'''DICCIONARIOS**************************************************************
 📚 Un **diccionario** (dict) es una colección **no ordenada** de elementos que 
     almacena datos en **pares clave - valor**. Cada elemento tiene una **clave única**, 
     que se utiliza para acceder o modificar su valor correspondiente.  

 🧠  Características principales:
        🔹 Los datos se guardan en pares: **clave : valor**.  
        🔹 No permiten claves duplicadas.  
        🔹 Son **mutables**, es decir, pueden modificarse después de su creación.  
        🔹 Se definen con llaves `{}`.  
        🔹 Son ideales para representar objetos, registros o información estructurada.  

 💡  Importancia:
        - Permiten **organizar datos de manera estructurada**, como si fueran 
          minibases de datos dentro del programa.  
        - Facilitan el acceso rápido a información mediante el uso de claves.  
        - Son esenciales para manipular datos tabulares o jerárquicos.  
        - En **econometría o análisis energético**, pueden representar registros 
          de clientes, zonas, indicadores o resultados de modelos.  

 ✍️  Sintaxis:
'''
producto = {"id": 1, "nombre": "Laptop", "precio": 750}
producto.get("stock", 0)   # acceso con valor por defecto (0)

producto["stock"] = 10                 # 1) crea/actualiza
producto.update({"precio": 800})       # 2) actualiza varios
valor = producto.pop("stock", None)    # 3) quita y retorna (o None)
del producto["precio"]                 # 4) elimina clave (si existe)
producto.clear()                       # 5) limpia todo

#recorridos
for k in producto.keys():     # claves
    print(k)

for v in producto.values():   # valores
    print(v)

for k, v in producto.items(): # pares clave-valor
    print(k, v)
 
    
'''⚙️ Ejemplo: Registro de consumo energético de clientes'''    
# Diccionario de clientes con sus datos
clientes = {
    "C001": {"nombre": "Ana Pérez", "zona": "Norte", "consumo_kwh": 450, "tarifa": 0.18},
    "C002": {"nombre": "Carlos Díaz", "zona": "Sur", "consumo_kwh": 620, "tarifa": 0.16},
    "C003": {"nombre": "Elena López", "zona": "Centro", "consumo_kwh": 300, "tarifa": 0.20}
}

# Calcular y agregar el monto a pagar por cliente
for codigo, datos in clientes.items():
    monto = datos["consumo_kwh"] * datos["tarifa"]
    clientes[codigo]["monto_pagar"] = monto

# Mostrar resultados
for codigo, datos in clientes.items():
    print(f"Cliente {codigo}: {datos['nombre']} | Zona: {datos['zona']} | Monto a pagar: ${datos['monto_pagar']:.2f}")

'''SALIDA
Cliente C001: Ana Pérez | Zona: Norte | Monto a pagar: $81.00
Cliente C002: Carlos Díaz | Zona: Sur | Monto a pagar: $99.20
Cliente C003: Elena López | Zona: Centro | Monto a pagar: $60.00

# len
len([1, 2, 3])             # 3

"Mouse" in ["Laptop", "Mouse"]      # True

d = {"x": 1, "y": 2}
del d["x"]                  # elimina clave x

l = [10, 20, 30]
del l[1]                    # elimina elemento en índice 1 -> [10, 30]

# update (diccionarios)
config = {"modo": "prod", "retry": 3}
config.update({"retry": 5, "cache": True})

# pop
val = config.pop("retry", 0) # 5; si no existe retorna 0

# clear
config.clear()              # {}

#Iteración con for y while
#for sobre lista/tupla/set
items = ["Laptop", "Mouse", "Teclado"]

for item in items:
    print(item)

# con índice
for i, item in enumerate(items, start=0):
    print(i, item)
    
#for sobre diccionario
producto = {"id": 1, "nombre": "Laptop", "precio": 750}

for clave in producto:  # equivalente a producto.keys()
    print(clave)

for clave, valor in producto.items():
    print(clave, valor)


#while controlado por condicion indice
numeros = [10, 20, 30, 40]
i = 0
while i < len(numeros):
    print(numeros[i])
    i += 1
